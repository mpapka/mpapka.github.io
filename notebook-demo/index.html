notebook demo

Runnable Python in the browser, powered by marimo.

This page is not linked from the site. It is a sandbox for trying runnable notebooks: the Python below runs in your browser through marimo and WebAssembly, with nothing installed and no server behind it. The first run takes a few seconds while Python and the packages load.

Notebooks are reactive. Move a slider and every cell that depends on it reruns. You can also edit the code in place.

<h1>Amdahl's law</h1>
<div class="uic-objectives">
  <strong>Objectives</strong>See why the serial part of a program caps its speedup, no matter how many processors you add.
</div>

import marimo as mo
import numpy as np
import matplotlib.pyplot as plt

parallel = mo.ui.slider(0.50, 0.99, step=0.01, value=0.95, label="parallel fraction")
max_procs = mo.ui.slider(2, 14, value=10, label="processors (2^n)")
mo.vstack([parallel, max_procs])

procs = 2 ** np.arange(0, max_procs.value + 1)
speedup = 1 / ((1 - parallel.value) + parallel.value / procs)

fig, ax = plt.subplots(figsize=(7, 3.5))
ax.semilogx(procs, speedup, marker="o", base=2, color="#001E62")
ax.axhline(1 / (1 - parallel.value), linestyle="--", color="#D50032")
ax.set_xlabel("processors")
ax.set_ylabel("speedup")
ax.set_title(f"{parallel.value * 100:.0f} percent parallel: speedup can never exceed {1 / (1 - parallel.value):.0f}x")
ax.grid(True, alpha=0.3)
fig

  what does the last 1 in 100 cost you?
</div>

<h1>Polaris right now</h1>
<div class="uic-objectives">
  <strong>Objectives</strong>Pull the live scheduler feed behind
  <a href="https://status.alcf.anl.gov/#/polaris" target="_blank">status.alcf.anl.gov</a> straight into the notebook, then slice it: how busy is the
  machine, which queues are backed up, and how long are jobs waiting?
</div>

import marimo as mo
import pandas as pd
import matplotlib.pyplot as plt
import json, time
from pyodide.http import pyfetch

STATUS_URL = "https://status.alcf.anl.gov/polaris/activity.json"
NAVY, RED, BLUE, GRAY = "#001E62", "#D50032", "#00B5E2", "#B8C0CC"

refresh = mo.ui.button(label="Fetch latest", value=0, on_click=lambda count: count + 1)
refresh

# Rerunning this cell (the button above) refetches the feed; everything below follows.
refresh.value
_response = await pyfetch(STATUS_URL)
status = json.loads(await _response.string())
minutes_old = (time.time() - status["updated"]) / 60
mo.md(f"Fetch number {refresh.value + 1}: the feed was updated **{minutes_old:.0f} minutes ago**.")

node_states = pd.Series([n["state"] for n in status["nodeinfo"].values()]).value_counts()
compute_nodes = sum(1 for name in status["nodeinfo"] if not name.startswith("polaris-login"))
busy_nodes = sum(job["nodes"] for job in status["running"])

mo.hstack([
    mo.stat(value=f"{busy_nodes} of {compute_nodes}", label="nodes running jobs",
            caption=f"{busy_nodes / compute_nodes * 100:.0f} percent of the machine"),
    mo.stat(value=str(len(status["running"])), label="jobs running"),
    mo.stat(value=str(len(status["queued"])), label="jobs waiting"),
    mo.stat(value=str(sum(int(r["resv_nodect"]) for r in status["reservation"])), label="nodes reserved"),
], justify="space-around")

_fig, _ax = plt.subplots(figsize=(7, 2.6))
_ax.barh(node_states.index[::-1], node_states.values[::-1], color=NAVY)
for _y, _count in enumerate(node_states.values[::-1]):
    _ax.text(_count + 4, _y, str(_count), va="center")
_ax.set_xlabel("nodes")
_ax.set_title("What every node is doing")
_ax.spines[["top", "right"]].set_visible(False)
_fig.tight_layout()
_fig

_now = time.time()
jobs = pd.DataFrame(
    [{"state": "running", "queue": j["queue"], "project": j["project"], "nodes": j["nodes"],
      "walltime (h)": round(j["walltime"] / 60, 1), "hours waiting": None,
      "hours running": round((_now - int(j["starttime"])) / 3600, 1)} for j in status["running"]]
    + [{"state": "queued", "queue": j["queue"], "project": j["project"], "nodes": j["nodes"],
        "walltime (h)": round(j["walltime"] / 60, 1),
        "hours waiting": round((_now - j["submittime"]) / 3600, 1), "hours running": None} for j in status["queued"]]
)
queue_pick = mo.ui.dropdown(options=["all queues"] + sorted(jobs["queue"].unique()), value="all queues", label="Queue")
queue_pick

picked = jobs if queue_pick.value == "all queues" else jobs[jobs["queue"] == queue_pick.value]

_by_queue = picked.pivot_table(index="queue", columns="state", values="nodes", aggfunc="sum", fill_value=0)
_by_queue = _by_queue.reindex(columns=["running", "queued"], fill_value=0).sort_values("queued")
_fig, _ax = plt.subplots(figsize=(7, max(2.2, 0.35 * len(_by_queue) + 1)))
_ax.barh(_by_queue.index, _by_queue["running"], color=NAVY, label="running")
_ax.barh(_by_queue.index, _by_queue["queued"], left=_by_queue["running"], color=RED, label="waiting")
_ax.set_xlabel("nodes")
_ax.set_title("Nodes running and waiting, by queue")
_ax.legend(frameon=False, loc="upper center", bbox_to_anchor=(0.5, -0.18), ncol=2)
_ax.spines[["top", "right"]].set_visible(False)
_fig.tight_layout()
_fig

# A few jobs sit for weeks; fold everything past two weeks into the last bin.
_days = picked["hours waiting"].dropna() / 24
_fig, _ax = plt.subplots(figsize=(7, 2.8))
_ax.hist(_days.clip(upper=14), bins=28, range=(0, 14), color=BLUE, edgecolor="white")
_ax.axvline(_days.median(), color=RED, linestyle="--")
_ax.set_xlabel("days since submission (14 = two weeks or more)")
_ax.set_ylabel("queued jobs")
_ax.set_title(f"How long queued jobs have waited (median {_days.median():.1f} days)")
_ax.spines[["top", "right"]].set_visible(False)
_fig.tight_layout()
_fig

mo.ui.table(picked.sort_values(["state", "nodes"], ascending=[False, False]), page_size=10, selection=None)
