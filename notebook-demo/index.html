notebook demo

Runnable Python in the browser, powered by marimo.

This page is not linked from the site. It is a sandbox for trying runnable notebooks: the Python below runs in your browser through marimo and WebAssembly, with nothing installed and no server behind it. The first run takes a few seconds while Python and the packages load.

Notebooks are reactive. Move a slider and every cell that depends on it reruns. You can also edit the code in place.

A second prototype serves the same demos from this site instead of marimo.app, which lets the UIC colors reach inside the widgets too.

<h1>Amdahl's law</h1>
<div class="uic-objectives">
  <strong>Objectives</strong>See why the serial part of a program caps its speedup, no matter how many processors you add.
</div>

import marimo as mo
import numpy as np
import matplotlib.pyplot as plt

parallel = mo.ui.slider(0.50, 0.99, step=0.01, value=0.95, label="parallel fraction")
max_procs = mo.ui.slider(2, 14, value=10, label="processors (2^n)")
mo.vstack([parallel, max_procs])

procs = 2 ** np.arange(0, max_procs.value + 1)
speedup = 1 / ((1 - parallel.value) + parallel.value / procs)

fig, ax = plt.subplots(figsize=(7, 3.5))
ax.semilogx(procs, speedup, marker="o", base=2, color="#001E62")
ax.axhline(1 / (1 - parallel.value), linestyle="--", color="#D50032")
ax.set_xlabel("processors")
ax.set_ylabel("speedup")
ax.set_title(f"{parallel.value * 100:.0f} percent parallel: speedup can never exceed {1 / (1 - parallel.value):.0f}x")
ax.grid(True, alpha=0.3)
fig

<h1>ALCF machines right now</h1>
<div class="uic-objectives">
  <strong>Objectives</strong>Pull the live scheduler feeds behind
  <a href="https://status.alcf.anl.gov/#/polaris" target="_blank">status.alcf.anl.gov</a> straight into the notebook, pick a machine, then slice it:
  how busy is it, where are the jobs sitting on the hardware, which queues are backed up, and how long are jobs waiting?
</div>

import marimo as mo
import pandas as pd
import matplotlib.pyplot as plt
import altair as alt
import json, time
from pyodide.http import pyfetch

alt.data_transformers.disable_max_rows()  # Aurora alone has over 10,000 nodes
NAVY, NAVY_SOFT, RED = "#001E62", "#1B3A7A", "#D50032"

machine = mo.ui.dropdown(options=["Polaris", "Aurora", "Crux", "Sophia"], value="Polaris", label="Machine")
refresh = mo.ui.button(label="Fetch latest", value=0, on_click=lambda count: count + 1)
mo.hstack([machine, refresh], justify="start", gap=1.5)

# Changing the machine or pressing the button reruns this cell; everything below follows.
refresh.value
_response = await pyfetch(f"https://status.alcf.anl.gov/{machine.value.lower()}/activity.json")
status = json.loads(await _response.string())
minutes_old = (time.time() - status["updated"]) / 60
mo.md(f"**{machine.value}**, fetch number {refresh.value + 1}: the feed was updated **{minutes_old:.0f} minutes ago**.")

node_states = pd.Series([n["state"] for n in status["nodeinfo"].values()]).value_counts()
_compute = {name for name in status["nodeinfo"] if "login" not in name}
compute_nodes = len(_compute)
# Count nodes, not job sizes: on Sophia several small jobs can share one DGX node.
busy_nodes = len(_compute & {node for job in status["running"] for node in job["location"]})

mo.hstack([
    mo.stat(value=f"{busy_nodes} of {compute_nodes}", label="nodes running jobs",
            caption=f"{busy_nodes / compute_nodes * 100:.0f} percent of the machine"),
    mo.stat(value=str(len(status["running"])), label="jobs running"),
    mo.stat(value=str(len(status["queued"])), label="jobs waiting"),
    mo.stat(value=str(sum(int(r["resv_nodect"]) for r in status["reservation"])), label="nodes reserved"),
], justify="space-around")

_fig, _ax = plt.subplots(figsize=(7, 2.6))
_ax.barh(node_states.index[::-1], node_states.values[::-1], color=NAVY)
for _y, _count in enumerate(node_states.values[::-1]):
    _ax.text(_count + node_states.max() * 0.01, _y, str(_count), va="center")
_ax.set_xlabel("nodes")
_ax.set_title("What every node is doing")
_ax.spines[["top", "right"]].set_visible(False)
_fig.tight_layout()
_fig

_now = time.time()
jobs = pd.DataFrame(
    [{"state": "running", "queue": j["queue"], "project": j["project"], "nodes": j["nodes"],
      "walltime (h)": round(j["walltime"] / 60, 1), "hours waiting": None,
      "hours running": round((_now - int(j["starttime"])) / 3600, 1)} for j in status["running"]]
    + [{"state": "queued", "queue": j["queue"], "project": j["project"], "nodes": j["nodes"],
        "walltime (h)": round(j["walltime"] / 60, 1),
        "hours waiting": round((_now - j["submittime"]) / 3600, 1), "hours running": None} for j in status["queued"]]
)
queue_pick = mo.ui.dropdown(options=["all queues"] + sorted(jobs["queue"].unique()), value="all queues", label="Queue")
queue_pick

# Machine layouts ported from the status page's own drawing code. Each returns
# (node, gpu, x, y, group) boxes with y growing upward; group is the rack label.
def polaris_layout():
    # Rows X30xx, X31xx, X32xx top to bottom; rack 1 on the right (the lower two rows have 12
    # racks); slots 1, 7, ... 37 from the bottom; two nodes per slot, b1 on the left.
    cells = []
    for row in range(3):
        for rack in range(1, 13 if row else 17):
            name = f"x{3000 + row * 100 + rack}"
            for k, slot in enumerate(range(1, 38, 6)):
                for blade in (0, 1):
                    cells.append((f"{name}c0s{slot}b{blade}n0", None, (16 - rack) * 2.6 + 1 - blade, -row * 9.0 + k, name.upper()))
    return cells

def aurora_layout():
    # Eight rows going up, 21 racks per row in groups of three; each rack is two halves of
    # four chassis (even chassis on the left) by eight nodes.
    cells = []
    for row in range(8):
        for rack in range(21):
            name = f"x4{row}{rack:02d}"
            _group_i, _in_group = divmod(rack, 3)  # no modulo sign: marimo.app drops code containing one
            x0 = _group_i * 54.0 + _in_group * 17.0
            for half in (0, 1):
                for k in range(4):
                    for s in range(8):
                        cells.append((f"{name}c{half + 2 * k}s{s}b0n0", None, x0 + half * 8.4 + s, row * 6.0 + k, f"ROW {row}"))
    return cells

def crux_layout():
    # One rack: two halves of four chassis, eight blades per chassis, four nodes per blade.
    cells = []
    for half in (0, 1):
        for k in range(4):
            for blade in range(8):
                for n, suffix in enumerate(("b0n0", "b0n1", "b1n0", "b1n1")):
                    cells.append((f"x1000c{half + 2 * k}s{blade}{suffix}", None, half * 8.6 + blade, k * 4.6 + n, f"chassis {half + 2 * k}"))
    return cells

def sophia_layout():
    # Drawn per GPU: racks 1-3 and 5-7, four DGX nodes per rack, two rows of four GPUs each.
    cells = []
    for r, label in enumerate((1, 2, 3, 5, 6, 7)):
        for k in range(4):
            for gpu in range(8):
                cells.append((f"sophia-gpu-{r * 4 + k + 1:02d}", gpu, r * 5.0 + divmod(gpu, 4)[1], k * 2.6 + divmod(gpu, 4)[0], f"rack {label}"))
    return cells

LAYOUTS = {"Polaris": polaris_layout, "Aurora": aurora_layout, "Crux": crux_layout, "Sophia": sophia_layout}
# Sized for the notebook column; Aurora is too big for that, so it scrolls sideways.
MAP_SIZE = {"Polaris": (600, 360), "Aurora": (2000, 560), "Crux": (420, 480), "Sophia": (600, 300)}

# The machine map: every box is a node (a GPU on Sophia), colored by the job running on it.
# Hover for details, click a box to select its whole job; the queue picker fades other queues.
_jobs_on = {}
for _job in status["running"]:
    for _node in _job["location"]:
        _jobs_on.setdefault(_node, []).append(_job)

_rows = []
for _node, _gpu, _x, _y, _group in LAYOUTS[machine.value]():
    if _node not in status["nodeinfo"]:
        continue  # a few layout positions are empty racks
    _state = status["nodeinfo"][_node]["state"]
    _here = _jobs_on.get(_node, [])
    if _gpu is not None:
        # Sophia jobs name a DGX node and a GPU count; fill that node's GPUs in order, as the status page does.
        _slots = [j for j in _here for _ in range(int(j["schedselect"][0].get("ngpus", 8)))]
        _job = _slots[_gpu] if _gpu < len(_slots) else None
    else:
        _job = _here[0] if _here else None
    if _job:
        _fill = _job["color"]
        _alpha = 0.12 if queue_pick.value not in ("all queues", _job["queue"]) else 0.95
    else:
        _fill = "#1F2933" if ("down" in _state or "offline" in _state) else "#C9D3E8" if "resv" in _state else "#E6E9EF"
        _alpha = 1.0
    _rows.append({
        "x": _x, "x2": _x + 0.9, "y": _y, "y2": _y + 0.85, "group": _group, "node": _node + ("" if _gpu is None else f" GPU {_gpu}"),
        "state": _state, "fill": _fill, "alpha": _alpha,
        "jobid": _job["jobid"].split(".")[0] if _job else None,
        "project": _job["project"] if _job else "", "queue": _job["queue"] if _job else "",
        "job nodes": _job["nodes"] if _job else None,
        "runtime": _job["runtimef"] if _job else "", "walltime": _job["walltimef"] if _job else "",
    })
map_cells = pd.DataFrame(_rows)

_labels = map_cells.groupby("group").agg(x=("x", "min"), xmax=("x2", "max"), y=("y2", "max"), ymin=("y", "min")).reset_index()
if machine.value == "Aurora":
    _labels = _labels.assign(x=_labels["x"] - 1.5, y=(_labels["y"] + _labels["ymin"]) / 2)  # row labels on the left
    _text = alt.Chart(_labels).mark_text(fontSize=9, align="right", color="#5B6573").encode(x="x:Q", y="y:Q", text="group:N")
else:
    _labels = _labels.assign(x=(_labels["x"] + _labels["xmax"]) / 2, y=_labels["y"] + 0.7)
    _text = alt.Chart(_labels).mark_text(fontSize=8, color="#5B6573").encode(x="x:Q", y="y:Q", text="group:N")

_pick = alt.selection_point(fields=["jobid"], name="job")
_boxes = alt.Chart(map_cells).mark_rect().encode(
    x=alt.X("x:Q", axis=None, scale=alt.Scale(zero=False, nice=False)), x2="x2:Q",
    y=alt.Y("y:Q", axis=None, scale=alt.Scale(zero=False, nice=False)), y2="y2:Q",
    color=alt.Color("fill:N", scale=None, legend=None),
    opacity=alt.condition(_pick, alt.Opacity("alpha:Q", scale=None, legend=None), alt.value(0.1)),
    tooltip=["node:N", "group:N", "state:N", "project:N", "queue:N", "job nodes:Q", "runtime:N", "walltime:N", "jobid:N"],
).add_params(_pick)
_w, _h = MAP_SIZE[machine.value]
machine_map = mo.ui.altair_chart(
    alt.layer(_boxes, _text).properties(width=_w, height=_h, title=f"{machine.value}: every colored box is busy with a job").configure_view(stroke=None),
    chart_selection=False, legend_selection=False,
)
mo.vstack([machine_map, mo.md("Light gray is idle, pale blue reserved but idle, black offline or down. Hover a box for details; click one to select its job.")])

# Whatever you click on the map arrives here as rows of data.
# The click arrives as the selected job id; Vega reads ids as numbers, so compare as text.
_ids = [str(i) for i in machine_map.selections.get("job", {}).get("jobid", []) if i is not None]
if len(_ids) == 1:
    _job = next(j for j in status["running"] if j["jobid"].split(".")[0] == _ids[0])
    _groups = map_cells[map_cells["jobid"] == _ids[0]]["group"].value_counts()
    _panel = mo.callout(mo.md(
        f"**Job {_ids[0]}**: project **{_job['project']}** in the **{_job['queue']}** queue, "
        f"{_job['nodes']} nodes, running {_job['runtimef']} of a {_job['walltimef']} walltime. "
        f"It spans {len(_groups)} of the map's groups: {', '.join(f'{g} ({n})' for g, n in _groups.items())}."
    ), kind="info")
else:
    _panel = mo.md("*Click a colored box on the map to see its job here.*")
_panel

picked = jobs if queue_pick.value == "all queues" else jobs[jobs["queue"] == queue_pick.value]

_by_queue = picked.pivot_table(index="queue", columns="state", values="nodes", aggfunc="sum", fill_value=0)
_by_queue = _by_queue.reindex(columns=["running", "queued"], fill_value=0).sort_values("queued")
_fig, _ax = plt.subplots(figsize=(7, max(2.2, 0.35 * len(_by_queue) + 1)))
_ax.barh(_by_queue.index, _by_queue["running"], color=NAVY, label="running")
_ax.barh(_by_queue.index, _by_queue["queued"], left=_by_queue["running"], color=RED, label="waiting")
_ax.set_xlabel("nodes")
_ax.set_title("Nodes running and waiting, by queue")
_ax.legend(frameon=False, loc="upper center", bbox_to_anchor=(0.5, -0.18), ncol=2)
_ax.spines[["top", "right"]].set_visible(False)
_fig.tight_layout()
_fig

# A few jobs sit for weeks; fold everything past two weeks into the last bin.
_days = picked["hours waiting"].dropna() / 24
_fig, _ax = plt.subplots(figsize=(7, 2.8))
_ax.hist(_days.clip(upper=14), bins=28, range=(0, 14), color=NAVY_SOFT, edgecolor="white")
_ax.axvline(_days.median(), color=RED, linestyle="--")
_ax.set_xlabel("days since submission (14 = two weeks or more)")
_ax.set_ylabel("queued jobs")
_ax.set_title(f"How long queued jobs have waited (median {_days.median():.1f} days)")
_ax.spines[["top", "right"]].set_visible(False)
_fig.tight_layout()
_fig

mo.ui.table(picked.sort_values(["state", "nodes"], ascending=[False, False]), page_size=10, selection=None)

<h1>Every widget</h1>
<div class="uic-objectives">
  <strong>Objectives</strong>See the marimo input widgets side by side; each name in the readout links to its page in the
  <a href="https://docs.marimo.io/api/inputs/" target="_blank">marimo widget reference</a>. Change any of them and the readout at the bottom updates,
  because it reads every widget's value.
</div>

import marimo as mo

slider = mo.ui.slider(0, 100, value=40, label="slider")
range_slider = mo.ui.range_slider(0, 100, value=[20, 70], label="range slider")
number = mo.ui.number(start=0, stop=10, step=0.5, value=3, label="number")
text = mo.ui.text(value="Polaris", label="text")
text_area = mo.ui.text_area(value="Notes go here", label="text area")
mo.vstack([slider, range_slider, number, text, text_area], gap=0.75)

dropdown = mo.ui.dropdown(["Aurora", "Polaris", "Crux", "Sophia"], value="Aurora", label="dropdown")
multiselect = mo.ui.multiselect(["CPU", "GPU", "memory", "network"], value=["GPU"], label="multiselect")
radio = mo.ui.radio(["batch", "interactive", "streaming"], value="batch", label="radio")
checkbox = mo.ui.checkbox(value=True, label="checkbox")
switch = mo.ui.switch(value=False, label="switch")
date = mo.ui.date(value="2026-12-02", label="date")
mo.hstack([mo.vstack([dropdown, multiselect, radio], gap=0.75), mo.vstack([checkbox, switch, date], gap=0.75)], justify="start", gap=3)

button = mo.ui.button(label="button (counts clicks)", value=0, on_click=lambda clicks: clicks + 1)
run_button = mo.ui.run_button(label="run button")
upload = mo.ui.file(label="file upload", filetypes=[".csv", ".txt"])
mo.hstack([button, run_button, upload], justify="start", gap=1)

table = mo.ui.table(
    [{"machine": "Aurora", "nodes": 10624}, {"machine": "Polaris", "nodes": 560},
     {"machine": "Crux", "nodes": 256}, {"machine": "Sophia", "nodes": 24}],
    label="table: tick rows to select them",
)
code = mo.ui.code_editor(value='print("hello from the code editor")', language="python", label="code editor")
mo.ui.tabs({"table": table, "code editor": code})

_picked_rows = ", ".join(row["machine"] for row in table.value) or "none"
_uploaded = upload.name() if upload.value else "none"
mo.md(f"""
